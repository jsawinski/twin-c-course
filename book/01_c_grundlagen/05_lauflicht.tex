Das Programm läuft dabei unendlich lange, bis es mit \texttt{Strg+C} abgebrochen wird.

\begin{code}
#include <stdio.h>

int main() {
    while(1) {
        for (int i = 0; i < 20; i++) {
            printf("\r");
            for (int j = 0; j < 20; j++) {
                if (i == j)
                    printf("*");
                else
                    printf(" ");
            }
        }
    }

    return 0;
}
\end{code}
